# Raspberry Pi üzerinde SSH debug komutları
# (Bu komutları Raspberry Pi'de yerel olarak çalıştırın)

# 1. SSH servisini kontrol et
sudo systemctl status ssh

# 2. SSH servisini başlat (gerekirse)
sudo systemctl start ssh
sudo systemctl enable ssh

# 3. SSH portunu kontrol et
sudo netstat -tlnp | grep :22

# 4. Firewall kontrol et
sudo ufw status

# 5. SSH config kontrol et
sudo nano /etc/ssh/sshd_config
# Kontrol edilecek satırlar:
# Port 22
# PermitRootLogin no
# PasswordAuthentication yes

# 6. SSH servisini yeniden başlat
sudo systemctl restart ssh

# 7. Local IP adresini kontrol et
ip addr show

# 8. Router/modem IP'sini kontrol et
ip route show | grep default

# 9. Dış IP'yi kontrol et (Pi üzerinden)
curl ifconfig.me

# 10. Test bağlantısı (local network içinden)
ssh pi@192.168.1.100
